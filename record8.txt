mysql> CREATE VIEW inv_pro AS SELECT customer.name, invoice_item.inv_id, invoice_item.pid, invoice_item.quantity FROM customer JOIN invoice_master ON
customer.id = invoice_master.cust_id JOIN invoice_item ON invoice_master.inv_id = invoice_item.inv_id;
